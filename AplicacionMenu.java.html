import java.util.ArrayList;
import java.util.List;
import java.util.NoSuchElementException;
import java.util.Scanner;

public class AplicacionMenu {

    private static final int SALIR = 7;

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        List<Integer> listaNumeros = new ArrayList<>();
        int opcion = 0;

        try {
            do {
                mostrarMenu();
                Integer lectura = leerEntero(scanner, "Seleccione una opcion: ");

                if (lectura == null) {
                    System.out.println("[ERROR] Debe ingresar un numero entero valido.");
                    opcion = 0; // evita conservar una opcion anterior
                } else {
                    opcion = lectura;
                    procesarOpcion(opcion, listaNumeros, scanner);
                }
                System.out.println();
            } while (opcion != SALIR);
        } catch (NoSuchElementException e) {
            // La entrada se cerro (Ctrl+D / Ctrl+Z): salimos sin lanzar excepcion
            System.out.println("\nEntrada finalizada. Saliendo de la aplicacion...");
        } finally {
            scanner.close();
        }
    }

    /**
     * Lee una linea completa y la convierte a entero.
     * Devuelve null si el texto no es un entero valido.
     * Usar nextLine() evita que queden caracteres pendientes en el bufer.
     */
    private static Integer leerEntero(Scanner scanner, String mensaje) {
        System.out.print(mensaje);
        String linea = scanner.nextLine().trim();
        try {
            return Integer.parseInt(linea);
        } catch (NumberFormatException e) {
            return null;
        }
    }

    // Muestra el menu de opciones en la consola
    public static void mostrarMenu() {
        System.out.println("=================================");
        System.out.println("        MENU DE OPCIONES         ");
        System.out.println("=================================");
        System.out.println("1.- Leer dato");
        System.out.println("2.- Mostrar numeros pares");
        System.out.println("3.- Mostrar los cuadrados de los valores");
        System.out.println("4.- Suma de los elementos de la lista");
        System.out.println("5.- Buscar un elemento de la lista");
        System.out.println("6.- Encontrar valor maximo");
        System.out.println("7.- Salir");
        System.out.println("=================================");
    }

    // Maneja la logica segun la opcion seleccionada
    public static void procesarOpcion(int opcion, List<Integer> lista, Scanner scanner) {
        switch (opcion) {
            case 1:
                leerDato(lista, scanner);
                break;
            case 2:
                mostrarPares(lista);
                break;
            case 3:
                mostrarCuadrados(lista);
                break;
            case 4:
                mostrarSuma(lista);
                break;
            case 5:
                buscarElemento(lista, scanner);
                break;
            case 6:
                mostrarMaximo(lista);
                break;
            case SALIR:
                System.out.println("Saliendo de la aplicacion...");
                break;
            default:
                System.out.println("[AVISO] Opcion no valida. Intente de nuevo.");
        }
    }

    // 1. Anade un entero a la lista
    public static void leerDato(List<Integer> lista, Scanner scanner) {
        Integer numero = leerEntero(scanner, "Ingrese un numero entero a la lista: ");
        if (numero == null) {
            System.out.println("[ERROR] Entrada invalida. No se agrego ningun valor.");
            return;
        }
        lista.add(numero);
        System.out.println("[OK] El numero " + numero + " fue agregado a la lista.");
    }

    // 2. Muestra los numeros pares de la lista
    public static void mostrarPares(List<Integer> lista) {
        if (lista.isEmpty()) {
            System.out.println("[AVISO] La lista esta vacia.");
            return;
        }

        StringBuilder pares = new StringBuilder();
        for (Integer num : lista) {
            if (num % 2 == 0) {
                pares.append(num).append(" ");
            }
        }

        if (pares.length() == 0) {
            System.out.println("No se encontraron numeros pares.");
        } else {
            System.out.println("Numeros pares en la lista: " + pares.toString().trim());
        }
    }

    // 3. Muestra el cuadrado de cada numero
    public static void mostrarCuadrados(List<Integer> lista) {
        if (lista.isEmpty()) {
            System.out.println("[AVISO] La lista esta vacia.");
            return;
        }

        System.out.println("Cuadrado de los valores de la lista:");
        for (Integer num : lista) {
            long cuadrado = (long) num * num; // long evita desbordamiento
            System.out.println("  " + num + " -> " + cuadrado);
        }
    }

    // 4. Calcula y muestra la suma total
    public static void mostrarSuma(List<Integer> lista) {
        if (lista.isEmpty()) {
            System.out.println("[AVISO] La lista esta vacia. La suma es 0.");
            return;
        }

        long suma = 0; // long evita desbordamiento
        for (Integer num : lista) {
            suma += num;
        }
        System.out.println("La suma de los elementos de la lista es: " + suma);
    }

    // 5. Busca un elemento y reporta todas las posiciones donde aparece
    public static void buscarElemento(List<Integer> lista, Scanner scanner) {
        if (lista.isEmpty()) {
            System.out.println("[AVISO] La lista esta vacia.");
            return;
        }

        Integer elemento = leerEntero(scanner, "Ingrese el elemento a buscar: ");
        if (elemento == null) {
            System.out.println("[ERROR] Entrada invalida.");
            return;
        }

        StringBuilder indices = new StringBuilder();
        for (int i = 0; i < lista.size(); i++) {
            if (lista.get(i).equals(elemento)) {
                if (indices.length() > 0) {
                    indices.append(", ");
                }
                indices.append(i);
            }
        }

        if (indices.length() > 0) {
            System.out.println("[OK] El numero " + elemento + " se encuentra en el/los indice(s): " + indices);
        } else {
            System.out.println("[ERROR] El numero " + elemento + " no se encuentra en la lista.");
        }
    }

    // 6. Determina y muestra el valor maximo
    public static void mostrarMaximo(List<Integer> lista) {
        if (lista.isEmpty()) {
            System.out.println("[AVISO] La lista esta vacia.");
            return;
        }

        int maximo = lista.get(0);
        for (Integer num : lista) {
            if (num > maximo) {
                maximo = num;
            }
        }
        System.out.println("El valor maximo de la lista es: " + maximo);
    }
}
